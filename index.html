-- ============================== -- MÁY BAY MULTIPLAYER -- ============================== create extension if not exists pgcrypto; create table if not exists public.aviator_rounds ( id uuid primary key default gen_random_uuid(), room text not null default 'QQ-2026', status text not null default 'waiting' check (status in ('waiting','running','crashed')), created_at timestamptz not null default now(), started_at timestamptz, crashed_at timestamptz, crash_multiplier numeric not null default 0, seed_hash text not null default encode(digest(gen_random_uuid()::text,'sha256'),'hex') ); create index if not exists aviator_rounds_room_created_idx on public.aviator_rounds(room, created_at desc); create table if not exists public.aviator_bets ( id uuid primary key default gen_random_uuid(), round_id uuid not null references public.aviator_rounds(id) on delete cascade, username text not null, bet numeric not null check (bet >= 20), status text not null default 'active' check (status in ('active','cashed','lost')), cashout_multiplier numeric, payout numeric not null default 0, current_multiplier numeric not null default 1, created_at timestamptz not null default now(), cashed_at timestamptz, unique(round_id, username) ); create index if not exists aviator_bets_round_idx on public.aviator_bets(round_id, created_at); alter table public.aviator_rounds enable row level security; alter table public.aviator_bets enable row level security; drop policy if exists "qq_round_select" on public.aviator_rounds; drop policy if exists "qq_bet_select" on public.aviator_bets; create policy "qq_round_select" on public.aviator_rounds for select to anon, authenticated using (true); create policy "qq_bet_select" on public.aviator_bets for select to anon, authenticated using (true); grant select on public.aviator_rounds, public.aviator_bets to anon, authenticated; -- Hệ số chung: tất cả trình duyệt tính cùng công thức từ started_at. create or replace function public.qq_aviator_multiplier(p_started_at timestamptz) returns numeric language sql stable as $$ select round((1 + power(greatest(0, extract(epoch from (clock_timestamp()-p_started_at))), 1.42) * 0.58)::numeric, 2); $$; -- Tạo phòng/ván. Trong 20 giây đầu mọi người cùng đặt cược; sau 20 giây tự chuyển RUNNING. create or replace function public.qq_get_or_start_aviator_round(p_room text default 'QQ-2026') returns public.aviator_rounds language plpgsql security definer set search_path = public as $$ declare r public.aviator_rounds; begin perform pg_advisory_xact_lock(hashtext('qq_aviator:'||coalesce(p_room,'QQ-2026'))); select * into r from public.aviator_rounds where room=coalesce(p_room,'QQ-2026') and status in ('waiting','running') order by created_at desc limit 1; if not found then insert into public.aviator_rounds(room,status,crash_multiplier) values(coalesce(p_room,'QQ-2026'),'waiting',greatest(1.01,round((0.97/greatest(random(),0.00001))::numeric,2))) returning * into r; elsif r.status='waiting' and now() >= r.created_at + interval '20 seconds' then update public.aviator_rounds set status='running', started_at=now() where id=r.id returning * into r; end if; return r; end; $$; grant execute on function public.qq_get_or_start_aviator_round(text) to anon, authenticated; -- Atomic đặt cược: trừ điểm online rồi ghi cược. create or replace function public.qq_place_aviator_bet( p_room text, p_round_id uuid, p_username text, p_bet numeric ) returns jsonb language plpgsql security definer set search_path = public as $$ declare r public.aviator_rounds; w public.user_wallets; b public.aviator_bets; begin if coalesce(trim(p_username),'')='' then return jsonb_build_object('ok',false,'message','Chưa đăng nhập.'); end if; select * into r from public.aviator_rounds where id=p_round_id and room=p_room for update; if not found or r.status <> 'waiting' then return jsonb_build_object('ok',false,'message','Ván đã bắt đầu hoặc kết thúc.'); end if; if p_bet < 20 then return jsonb_build_object('ok',false,'message','Tối thiểu 20 điểm.'); end if; select * into b from public.aviator_bets where round_id=r.id and lower(username)=lower(p_username) limit 1; if found then return jsonb_build_object('ok',false,'message','Bạn đã cược trong ván này.'); end if; select * into w from public.user_wallets where lower(username)=lower(p_username) for update; if not found then insert into public.user_wallets(username,balance,points) values(p_username,0,0) returning * into w; end if; if w.points < p_bet then return jsonb_build_object('ok',false,'message','Không đủ điểm.'); end if; update public.user_wallets set points=points-p_bet, updated_at=now() where username=w.username; insert into public.aviator_bets(round_id,username,bet,status) values(r.id,p_username,p_bet,'active') returning * into b; return jsonb_build_object('ok',true,'id',b.id,'round_id',b.round_id,'username',b.username,'bet',b.bet,'status',b.status); end; $$; grant execute on function public.qq_place_aviator_bet(text,uuid,text,numeric) to anon, authenticated; -- Chốt điểm online theo thời gian server; nếu máy bay đã crash thì không cho chốt. create or replace function public.qq_cashout_aviator_bet(p_bet_id uuid, p_username text) returns jsonb language plpgsql security definer set search_path = public as $$ declare b public.aviator_bets; r public.aviator_rounds; w public.user_wallets; m numeric; payout numeric; begin select * into b from public.aviator_bets where id=p_bet_id and lower(username)=lower(p_username) for update; if not found then return jsonb_build_object('ok',false,'message','Không tìm thấy cược.'); end if; if b.status <> 'active' then return jsonb_build_object('ok',false,'message','Cược này đã được xử lý.'); end if; select * into r from public.aviator_rounds where id=b.round_id for update; if r.status <> 'running' or r.started_at is null then return jsonb_build_object('ok',false,'message','Ván chưa bay.'); end if; m:=least(public.qq_aviator_multiplier(r.started_at),r.crash_multiplier); if m >= r.crash_multiplier then update public.aviator_bets set status='lost', current_multiplier=r.crash_multiplier where id=b.id; return jsonb_build_object('ok',false,'message','Máy bay đã crash.'); end if; payout:=floor(b.bet*m); update public.aviator_bets set status='cashed', cashout_multiplier=m, payout=payout, current_multiplier=m, cashed_at=now() where id=b.id; select * into w from public.user_wallets where lower(username)=lower(b.username) for update; if not found then insert into public.user_wallets(username,balance,points) values(b.username,0,payout) returning * into w; else update public.user_wallets set points=points+payout, updated_at=now() where username=w.username; end if; return jsonb_build_object('ok',true,'multiplier',m,'payout',payout,'bet',b.bet); end; $$; grant execute on function public.qq_cashout_aviator_bet(uuid,text) to anon, authenticated; -- Kết thúc ván đúng một lần; mọi cược ACTIVE chưa chốt thành LOST. create or replace function public.qq_crash_aviator_round(p_round_id uuid) returns public.aviator_rounds language plpgsql security definer set search_path = public as $$ declare r public.aviator_rounds; begin select * into r from public.aviator_rounds where id=p_round_id for update; if not found then return null; end if; if r.status='running' and public.qq_aviator_multiplier(r.started_at) >= r.crash_multiplier then update public.aviator_rounds set status='crashed', crashed_at=now() where id=r.id returning * into r; update public.aviator_bets set status='lost', current_multiplier=r.crash_multiplier where round_id=r.id and status='active'; end if; return r; end; $$; grant execute on function public.qq_crash_aviator_round(uuid) to anon, authenticated;
