return new; end; $$; drop trigger if exists qq_wallet_updated_at on public.user_wallets; create trigger qq_wallet_updated_at before update on public.user_wallets for each row execute function public.qq_wallet_touch_updated_at(); -- ========================= -- 2) MÁY BAY — VÁN -- ========================= create table if not exists public.aviator_rounds ( id uuid primary key default gen_random_uuid(), room text not null default 'QQ-2026', round_no bigint generated by default as identity, status text not null default 'waiting' check (status in ('waiting','running','crashed')), created_at timestamptz not null default now(), started_at timestamptz, crashed_at timestamptz, crash_multiplier numeric, unique(room, round_no) ); create index if not exists aviator_rounds_room_status_idx on public.aviator_rounds(room, status, created_at desc); -- ========================= -- 3) MÁY BAY — CƯỢC -- ========================= create table if not exists public.aviator_bets ( id uuid primary key default gen_random_uuid(), round_id uuid not null references public.aviator_rounds(id) on delete cascade, username text not null, bet numeric not null check (bet > 0), status text not null default 'active' check (status in ('active','cashed','lost')), created_at timestamptz not null default now(), cashed_at timestamptz, cash_multiplier numeric, payout numeric not null default 0, unique(round_id, username) ); create index if not exists aviator_bets_round_idx on public.aviator_bets(round_id, created_at); create index if not exists aviator_bets_user_idx on public.aviator_bets(lower(username), created_at desc); exception when duplicate_object then null; end; begin alter publication supabase_realtime add table public.user_wallets; exception when duplicate_object then null; end; end $$; -- ========================= -- 5) RLS MÁY BAY -- ========================= alter table public.aviator_rounds enable row level security; alter table public.aviator_bets enable row level security; drop policy if exists "qq_round_select" on public.aviator_rounds; drop policy if exists "qq_round_insert" on public.aviator_rounds; drop policy if exists "qq_round_update" on public.aviator_rounds; create policy "qq_round_select" on public.aviator_rounds for select to anon, authenticated using (true); create policy "qq_round_insert" on public.aviator_rounds for insert to anon, authenticated with check (true); create policy "qq_round_update" on public.aviator_rounds for update to anon, authenticated using (true) with check (true); drop policy if exists "qq_bet_select" on public.aviator_bets; drop policy if exists "qq_bet_insert" on public.aviator_bets; drop policy if exists "qq_bet_update" on public.aviator_bets; create policy "qq_bet_select" on public.aviator_bets for select to anon, authenticated using (true); create policy "qq_bet_insert" on public.aviator_bets for insert to anon, authenticated with check (true); create policy "qq_bet_update" on public.aviator_bets for update to anon, authenticated using (true) with check (true); -- ========================= -- 6) TẠO / LẤY VÁN HIỆN TẠI -- ========================= create or replace function public.qq_get_or_start_aviator_round( p_room text default 'QQ-2026' ) returns jsonb language plpgsql security definer set search_path = public as $$ declare r public.aviator_rounds; begin select * into r from public.aviator_rounds where room = p_room and status in ('waiting','running') order by created_at desc limit 1 for update; if not found then insert into public.aviator_rounds(room,status) values(p_room,'waiting') returning * into r; end if; -- Nếu ván chờ đã quá 30 giây thì bắt đầu. if r.status = 'waiting' and now() >= r.created_at + interval '30 seconds' then update public.aviator_rounds set status='running', started_at=coalesce(started_at,now()) where id=r.id returning * into r; end if; return jsonb_build_object( 'ok', true, 'round', to_jsonb(r) ); end; $$; -- ========================= -- 7) HỆ SỐ MÁY BAY -- Dựa trên thời gian server, mọi người cùng 1 hệ số. -- ========================= create or replace function public.qq_aviator_multiplier( p_started_at timestamptz ) returns numeric language sql stable as $$ select round( greatest( 1.00, exp(greatest(0, extract(epoch from (now() - p_started_at))) * 0.105) )::numeric, 2 ); $$; -- ========================= -- 8) ĐẶT CƯỢC -- Trừ tiền online ngay khi cược. -- ========================= create or replace function public.qq_place_aviator_bet( p_round_id uuid, p_username text, p_bet numeric ) returns jsonb language plpgsql security definer set search_path = public as $$ declare r public.aviator_rounds; w public.user_wallets; b public.aviator_bets; begin if p_username is null or trim(p_username) = '' then return jsonb_build_object('ok',false,'message','Thiếu username.'); end if; if p_bet is null or p_bet <= 0 then return jsonb_build_object('ok',false,'message','Tiền cược không hợp lệ.'); end if; select * into r from public.aviator_rounds where id=p_round_id for update; if not found then return jsonb_build_object('ok',false,'message','Không tìm thấy ván.'); end if; if r.status <> 'waiting' then return jsonb_build_object('ok',false,'message','Ván đã bắt đầu, không thể cược.'); end if; if now() >= r.created_at + interval '30 seconds' then update public.aviator_rounds set status='running', started_at=coalesce(started_at,now()) where id=r.id; return jsonb_build_object('ok',false,'message','Hết thời gian cược.'); end if; if exists ( select 1 from public.aviator_bets where round_id=p_round_id and lower(username)=lower(trim(p_username)) ) then return jsonb_build_object('ok',false,'message','Tài khoản đã cược trong ván này.'); end if; select * into w from public.user_wallets where lower(username)=lower(trim(p_username)) for update; if not found then insert into public.user_wallets(username,balance,points) values(trim(p_username),0,0) returning * into w; end if; if w.balance < p_bet then return jsonb_build_object('ok',false,'message','Không đủ số dư.'); end if; update public.user_wallets set balance=balance-p_bet, updated_at=now() where username=w.username; insert into public.aviator_bets(round_id,username,bet,status) values(p_round_id,w.username,p_bet,'active') returning * into b; return jsonb_build_object( 'ok',true, 'message','Đặt cược thành công.', 'bet',to_jsonb(b) ); end; $$; -- ========================= -- 9) CHỐT CƯỢC -- Trả bet * multiplier. -- ========================= create or replace function public.qq_cashout_aviator_bet( p_bet_id uuid, p_username text ) returns jsonb language plpgsql security definer set search_path = public as $$ declare b public.aviator_bets; r public.aviator_rounds; w public.user_wallets; m numeric; payout_amount numeric; begin select * into b from public.aviator_bets where id=p_bet_id and lower(username)=lower(trim(p_username)) for update; if not found then return jsonb_build_object('ok',false,'message','Không tìm thấy cược.'); end if; if b.status <> 'active' then return jsonb_build_object('ok',false,'message','Cược này đã được xử lý.'); end if; select * into r from public.aviator_rounds where id=b.round_id for update; if r.status <> 'running' or r.started_at is null then return jsonb_build_object('ok',false,'message','Ván chưa bay.'); end if; m := public.qq_aviator_multiplier(r.started_at); if r.crash_multiplier is not null and m >= r.crash_multiplier then update public.aviator_bets set status='lost', payout=0 where id=b.id; return jsonb_build_object('ok',false,'message','Máy bay đã nổ.'); end if; payout_amount := floor(b.bet*m); update public.aviator_bets set status='cashed', cash_multiplier=m, cashed_at=now(), payout=payout_amount where id=b.id; select * into w from public.user_wallets where lower(username)=lower(trim(p_username)) for update; if not found then insert into public.user_wallets(username,balance,points) values(trim(p_username),payout_amount,0) returning * into w; else update public.user_wallets set balance=balance+payout_amount, updated_at=now() where username=w.username; end if; return jsonb_build_object( 'ok',true, 'message','Chốt thành công.', 'multiplier',m, 'payout',payout_amount ); end; $$; -- ========================= -- 10) NỔ VÁN -- Server đặt crash multiplier và xử lý cược còn active. -- ========================= create or replace function public.qq_crash_aviator_round( p_round_id uuid ) returns jsonb language plpgsql security definer set search_path = public as $$ declare r public.aviator_rounds; begin select * into r from public.aviator_rounds where id=p_round_id for update; if not found then return jsonb_build_object('ok',false,'message','Không tìm thấy ván.'); end if; if r.status <> 'running' then return jsonb_build_object('ok',false,'message','Ván chưa chạy.'); end if; -- Crash demo trong khoảng 2.20x - 4.40x. update public.aviator_rounds set crash_multiplier = round((2.20 + random()*2.20)::numeric,2), status='crashed', crashed_at=now() where id=r.id returning * into r; update public.aviator_bets set status='lost', payout=0 where round_id=r.id and status='active'; return jsonb_build_object( 'ok',true, 'crash_multiplier',r.crash_multiplier ); end; $$; -- ========================= -- 11) QUYỀN GỌI FUNCTION -- ========================= grant execute on function public.qq_get_or_start_aviator_round(text) to anon, authenticated; grant execute on function public.qq_crash_aviator_round(uuid) to anon, authenticated; -- ========================= -- 12) TẠO VÍ MẪU NẾU CẦN -- Không tự cộng tiền. -- ========================= insert into public.user_wallets(username,balance,points) values ('ADMIN_SocNau03',0,0) on conflict (username) do nothing;
